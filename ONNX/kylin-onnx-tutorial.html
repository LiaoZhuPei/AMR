构建 AArch64 模拟容器

创建一个模拟飞腾CPU的Docker容器，在里面安装ONNX Runtime

💡 为什么用Docker而不是直接WSL2

WSL2的内核还是x86_64。要真正模拟AArch64的运行时环境（确认ONNX包能在飞腾上跑），需要用Docker拉取arm64v8镜像，配合QEMU在x86机器上模拟执行。

// 2.1 创建工作目录

cd ~/kylin-deploy

Dockerfile（粘贴到文件）

编写 Python 推理服务

这是你提交给C++系统的核心模块——一个HTTP推理服务

// 4.1 主服务文件

在WSL中创建 ~/kylin-deploy/service/infer_server.py：

Python - infer_server.py 
"""
麒麟+飞腾 ONNX 推理服务
提供HTTP接口，供C++或其他系统调用
"""
import os, logging
import numpy as np
import onnxruntime as ort
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import List
import uvicorn

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("InferServer")

# ========== 配置（根据你的模型修改）==========
MODEL_PATH = os.environ.get("MODEL_PATH", "./model/model.onnx")
HOST = "0.0.0.0"
PORT = int(os.environ.get("PORT", 8000))
# ==============================================

app = FastAPI(title="ONNX推理服务", version="1.0")

# 全局Session（进程启动时加载一次，避免重复加载开销）
session: ort.InferenceSession = None
input_name: str = None
output_names: List[str] = None

class InferRequest(BaseModel):
    # data: 二维列表，shape=[batch_size, feature_dim]
    # 根据你的模型输入修改
    data: List[List[float]]

class InferResponse(BaseModel):
    result: List[List[float]]
    shape: List[int]

@app.on_event("startup")
async def startup():
    global session, input_name, output_names
    logger.info(f"加载模型: {MODEL_PATH}")
    
    # 只使用CPU（麒麟飞腾无GPU）
    opts = ort.SessionOptions()
    opts.inter_op_num_threads = 4
    opts.intra_op_num_threads = 4
    
    session = ort.InferenceSession(
        MODEL_PATH,
        sess_options=opts,
        providers=['CPUExecutionProvider']
    )
    input_name = session.get_inputs()[0].name
    output_names = [o.name for o in session.get_outputs()]
    logger.info(f"✓ 模型加载完成 | 输入: {input_name} | 输出: {output_names}")

@app.get("/")
def health():
    return {
        "status": "ok",
        "model": MODEL_PATH,
        "onnxruntime_version": ort.__version__
    }

@app.post("/infer", response_model=InferResponse)
async def infer(req: InferRequest):
    if session is None:
        raise HTTPException(503, "模型未加载")
    try:
        # 转换输入
        arr = np.array(req.data, dtype=np.float32)
        
        # 推理
        outputs = session.run(output_names, {input_name: arr})
        result = outputs[0]
        
        return InferResponse(
            result=result.tolist(),
            shape=list(result.shape)
        )
    except Exception as e:
        raise HTTPException(500, str(e))

if __name__ == "__main__":
    uvicorn.run(app, host=HOST, port=PORT, log_level="info")

// C++ 调用示例（cpp-httplib，仅需头文件）
C++ - infer_client.cpp 
// 依赖：https://github.com/yhirose/cpp-httplib (单头文件)
// 下载 httplib.h 放到同目录即可，无需编译任何库
#include "httplib.h"
#include <string>
#include <vector>
#include <iostream>
#include <sstream>

// 简单的JSON构建（避免引入json库）
std::string build_request(const std::vector<float>& data) {
    std::ostringstream ss;
    ss << "{\"data\": [[";
    for (size_t i = 0; i < data.size(); ++i) {
        if (i > 0) ss << ",";
        ss << data[i];
    }
    ss << "]]}";
    return ss.str();
}

// 调用推理服务
std::string call_infer(const std::vector<float>& input_data) {
    httplib::Client cli("localhost", 8000);
    cli.set_connection_timeout(5);  // 5秒超时
    cli.set_read_timeout(30);       // 推理超时
    
    std::string body = build_request(input_data);
    
    auto res = cli.Post("/infer", body, "application/json");
    
    if (!res || res->status != 200) {
        throw std::runtime_error("推理服务调用失败");
    }
    return res->body;  // JSON字符串，解析交给调用方
}

int main() {
    // 准备输入（128维float向量，根据你的模型修改）
    std::vector<float> input(128, 0.5f);
    
    try {
        std::string result = call_infer(input);
        std::cout << "推理结果: " << result << std::endl;
    } catch (const std::exception& e) {
        std::cerr << "错误: " << e.what() << std::endl;
        return 1;
    }
    return 0;
}
